resultado.dividendos = dividendos; resultado.resumenDivs = resumenDivs; // ═══ FIN TR_DATOS ═══ resultado.cierreAnterior = obtenerCierreAnterior(); resultado.historicoCierres = obtenerHistoricoCierres(); resultado.timestamp = new Date().toISOString(); resultado.ok = true; return ContentService .createTextOutput(JSON.stringify(resultado)) .setMimeType(ContentService.MimeType.JSON); } // ═══ RECÁLCULO AUTOMÁTICO DE GOOGLEFINANCE CADA 5 MINUTOS ═══ function forzarRecalculo() { var ss = SpreadsheetApp.getActiveSpreadsheet();
